# P5 — 工作流管理工具

> 依赖：P4（工作流执行引擎）
> 目标：Agent 可通过工具管理工作流（列表/启停/删除/立即执行/模板转正式）
> 原则：作为原子工具注册到 _REGISTRY，操作统一 workflows 表（P1 创建）

## 一、工具定义（★ R-013/R-014 修订版）

```python
# backend/services/agent_service/tools/atomic.py 末尾追加
# （list_workflows 和 manage_workflow 在 H 组原子工具）

# ───────────────────────────────────────────────────────────────
# H12. list_workflows — 列出当前用户可见工作流
# ───────────────────────────────────────────────────────────────
@tool(
    name="list_workflows",
    description=(
        "列出当前用户可见的所有工作流（user 自定义 + auto_template 累积 + 预置组合工具 preset）。"
        "用户问'我有哪些工作流'、'看 preset 工具'时调用。"
    ),
    parameters=None,
    group="atomic",
)
@with_owner
def list_workflows_tool(_owner_id: str) -> str:
    """返回工作流列表，按 source 分组"""
    from backend.services.agent_service.workflows_db import WorkflowsDB
    db = WorkflowsDB()
    # ★ R-013 修订：(owner=? AND source IN ('user','auto_template')) OR (source='preset')
    # 旧 SQL 只查 owner 自己的 + user/auto_template，会漏掉 preset
    items = db.query(
        """
        SELECT id, owner_id, name, description, source,
               schedule_cron, schedule_desc, enabled,
               hit_count, last_hit, last_run, run_count,
               is_composite_tool, created_at, updated_at
        FROM workflows
        WHERE (owner_id = ? AND source IN ('user', 'auto_template'))
           OR (source = 'preset')
        ORDER BY source ASC, updated_at DESC
        """,
        [_owner_id],
    )
    return json.dumps({
        "success": True,
        "data": items,
        "ui": {"type": "workflow_list", "data": {"items": items, "count": len(items)}},
    }, ensure_ascii=False, default=str)


# ───────────────────────────────────────────────────────────────
# H13. manage_workflow — 管理工作流（5 种 action）
# ───────────────────────────────────────────────────────────────
@tool(
    name="manage_workflow",
    description=(
        "管理工作流。action: enable/disable/delete/run_now/save_as_preset。"
        "enable/disable：开关。delete：删除（preset 不可删）。"
        "run_now：立即执行（仅 user/auto_template）。"
        "save_as_preset：将 auto_template 升级为 user 工作流。"
    ),
    parameters={
        "type": "object",
        "properties": {
            "workflow_id": {"type": "integer", "description": "工作流 ID"},
            "action": {"type": "string",
                       "enum": ["enable", "disable", "delete", "run_now", "save_as_preset"],
                       "description": "操作类型"},
        },
        "required": ["workflow_id", "action"],
    },
    group="atomic",
)
@with_owner
def manage_workflow_tool(workflow_id: int, action: str, _owner_id: str) -> str:
    from backend.services.agent_service.workflows_db import WorkflowsDB
    db = WorkflowsDB()
    wf = db.get_by_id(workflow_id)

    # 权限校验：工作流存在 + 仅本人可操作（system owner 可被 admin 改）
    if not wf:
        return json.dumps({"success": False, "error": "工作流不存在"}, ensure_ascii=False)
    if wf["owner_id"] != _owner_id and wf["owner_id"] != "system":
        return json.dumps({
            "success": False, "error": "无权操作他人工作流"
        }, ensure_ascii=False)

    if action == "enable":
        db.set_enabled(workflow_id, True)
        msg = "已启用"

    elif action == "disable":
        db.set_enabled(workflow_id, False)
        msg = "已停用"

    elif action == "delete":
        # ★ preset 不可删除（只能 disable）
        if wf["source"] == "preset":
            return json.dumps({
                "success": False,
                "error": "系统预置工作流不可删除，请使用 disable 停用"
            }, ensure_ascii=False)
        db.delete(workflow_id)
        msg = "已删除"

    elif action == "run_now":
        # 仅 user / auto_template 可手动触发
        if wf["source"] == "preset":
            return json.dumps({
                "success": False,
                "error": "预置组合工具由 Agent Function Calling 调用，无需手动 run_now"
            }, ensure_ascii=False)
        # 异步执行（不影响当前对话）
        import asyncio
        from backend.services.radar_service.scheduler import _execute_workflow_row
        try:
            loop = asyncio.get_running_loop()
            loop.create_task(_execute_workflow_row(wf))
        except RuntimeError:
            # 不在 event loop 中：用 ThreadPoolExecutor 兜底
            import concurrent.futures
            with concurrent.futures.ThreadPoolExecutor(max_workers=1) as ex:
                ex.submit(_run_workflow_sync, wf)
        msg = "已加入执行队列"

    elif action == "save_as_preset":
        # ★ R-014 修订：只允许 auto_template → user，禁止其他转换
        # （避免用户擅自把 user 升为 preset 污染全局）
        if wf["source"] != "auto_template":
            return json.dumps({
                "success": False,
                "error": "save_as_preset 只能将 auto_template 转为 user 工作流。"
                         "若需把 user 升为 preset，请联系 Admin。"
            }, ensure_ascii=False)
        db.update(workflow_id, source="user")
        msg = "已转为 user 工作流（可加 cron 调度）"

    else:
        return json.dumps({
            "success": False, "error": f"未知 action: {action}"
        }, ensure_ascii=False)

    return json.dumps({
        "success": True,
        "action": action,
        "workflow_id": workflow_id,
        "message": msg,
        "ui": {"type": "workflow_action_result", "data": {
            "workflow_id": workflow_id, "action": action, "message": msg,
        }},
    }, ensure_ascii=False)


def _run_workflow_sync(workflow: dict):
    """run_now 在非 event loop 上下文时的同步兜底"""
    import asyncio
    loop = asyncio.new_event_loop()
    try:
        from backend.services.radar_service.scheduler import _execute_workflow_row
        loop.run_until_complete(_execute_workflow_row(workflow))
    finally:
        loop.close()
```

## 三、验收标准

- [ ] `list_workflows` 返回当前用户的 user + auto_template + 全局 preset
- [ ] `manage_workflow(action="disable")` 停用后调度器不再触发
- [ ] `manage_workflow(action="delete")` 拒绝删除 source='preset' 的条目
- [ ] `manage_workflow(action="run_now")` 立即异步执行，不影响当前对话
- [ ] `manage_workflow(action="run_now")` 拒绝 preset（避免误改全局）
- [ ] **R-013**：`list_workflows` 返回 preset（即使是其他用户的 system 工作流，所有用户可见）
- [ ] **R-014**：`save_as_preset` 只能从 auto_template 转 user
- [ ] **R-014**：`save_as_preset` 拒绝 user → preset 的越权转换
- [ ] 用户 A 不能操作用户 B 的 user / auto_template 工作流（owner 校验生效）
- [ ] `manage_workflow` 返回 JSON 含 `action` + `workflow_id` + `message`，便于前端确认

## 四、典型对话流程

```
用户: "列出我的所有工作流"
Agent: 调用 list_workflows → 返回 [{preset...}, {auto_template...}, {user...}]

用户: "停用工作流 3"
Agent: 调用 manage_workflow(workflow_id=3, action="disable")
       → 校验：wf.owner_id == ctx.owner_id? → 是
       → db.set_enabled(3, False)
       → 返回 {"success": true, "message": "已停用"}

用户: "把工作流 5 转成定时任务，每天早上 8 点执行"
Agent: 调用 manage_workflow(workflow_id=5, action="save_as_preset")
       → 校验：wf.source == "auto_template"? → 是
       → db.update(5, source="user")
       → 然后通过其他工具（待补 P-extra）更新 schedule_cron="0 8 * * *"
```
