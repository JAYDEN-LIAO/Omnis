# P4 — 用户工作流（source='user'）+ cron 调度 + 取消

> 依赖：P1（统一 workflows 表 + execute_workflow_steps）、P2（phase_plan 的 mode=scheduled 输出）
> 目标：用户可通过自然语言创建定时工作流，系统自动按 cron 执行
> 原则：与 P1 共用 workflows 表和 execute_workflow_steps 执行器，source='user' 区分

## 一、存储模型

复用 P1 创建的 `workflows` 表，用户工作流 = `source='user'` 的行：

```sql
-- 查用户工作流
SELECT * FROM workflows WHERE owner_id = 'u_xxx' AND source = 'user';

-- 查待执行的定时任务
SELECT * FROM workflows WHERE source = 'user' AND schedule_cron != '' AND enabled = 1;
```

## 二、创建流程（★ R-010 修订：创建后不自动执行）

```
用户: "帮我设置一下，每天早上 9 点扫描理想汽车的负面舆情，发到我的邮箱"

Agent:
  phase_plan 输出:
    goal: "每天早上9点扫描理想汽车负面舆情"
    mode: "scheduled"
    schedule: {"cron": "0 9 * * *", "description": "每天早上9点"}
    steps: [搜索 → 过滤风险 → 聚类 → 情感分析 → 推送]

  P2 检测 mode == "scheduled"：
    1. _persist_workflow() → INSERT INTO workflows (source='user')
    2. emit_done_with_message() 回复用户
    3. return（不进入主循环）

  Agent 回复: "已创建定时任务（ID=1），每天早上 9 点自动执行。
              如需立即试运行，告诉我'立即执行工作流 1'。"
```

`_persist_workflow` 已在 P2 实现（P2 第十二节），这里不再重复。

## 三、执行引擎（★ R-011 修订：trace_id + Semaphore 限流）

```python
# backend/services/radar_service/scheduler.py 末尾新增

from datetime import datetime
import time
import json
import asyncio
import threading

_MAX_CONCURRENT_WORKFLOWS = 4
_WF_SEMAPHORE = asyncio.Semaphore(_MAX_CONCURRENT_WORKFLOWS)
_WF_CANCEL_EVENTS: dict = {}
_WF_CANCEL_LOCK = threading.Lock()


def _run_user_workflows():
    """每分钟检查一次，执行到期的用户工作流。

    ★ R-011 修订：入口 set_trace_id（与 _run_daily_summary 模式一致）；
                  Semaphore(4) 限流防止并发打爆。
    """
    # ★ R-011 关键：scheduler 跑在独立线程，ContextVar 不自动传播
    set_trace_id(generate_trace_id())
    try:
        loop = asyncio.new_event_loop()
        asyncio.set_event_loop(loop)
        try:
            loop.run_until_complete(_run_due_workflows())
        finally:
            loop.close()
    except Exception as e:
        logger.error(f"[Scheduler] _run_user_workflows 异常: {e}", exc_info=True)


async def _run_due_workflows():
    """遍历到期工作流，用 Semaphore 限流执行。"""
    from backend.services.agent_service.workflows_db import WorkflowsDB
    db = WorkflowsDB()
    rows = db.list_due_user_workflows()
    for row in rows:
        # ★ R-011 限流：每个工作流用 Semaphore 包裹，最多 4 个并发
        await _WF_SEMAPHORE.acquire()
        try:
            await _execute_workflow_row(row)
        finally:
            _WF_SEMAPHORE.release()


async def _execute_workflow_row(workflow: dict):
    """执行单条用户工作流。复用 P1 的 execute_workflow_steps。"""
    from backend.services.agent_service.workflows_db import execute_workflow_steps

    wf_id = workflow["id"]
    owner_id = workflow["owner_id"]
    steps = json.loads(workflow["steps_json"])

    cancel_event = asyncio.Event()
    with _WF_CANCEL_LOCK:
        _WF_CANCEL_EVENTS[wf_id] = cancel_event

    start = time.time()
    status = "success"
    error_msg = ""
    try:
        result_json = await execute_workflow_steps(
            steps=steps,
            initial_params={},
            owner_id=owner_id,
            cancel_event=cancel_event,
            workflow_id=wf_id,
            workflow_name=workflow["name"],
        )
        result = json.loads(result_json)
        duration = time.time() - start
        if result.get("error"):
            if result.get("message") == "用户取消":
                status = "cancelled"
            else:
                status = "failed"
            error_msg = result.get("error", "")
            logger.warning(
                f"[Workflow] wf={wf_id} {status}: "
                f"step={result.get('step')} err={error_msg[:100]}"
            )
        else:
            status = "success"
            logger.info(f"[Workflow] wf={wf_id} success: {duration:.1f}s")
            _push_workflow_result(owner_id, workflow["name"], result.get("results", {}))
    except Exception as e:
        status = "failed"
        error_msg = str(e)
        logger.error(f"[Workflow] wf={wf_id} 异常: {e}", exc_info=True)
    finally:
        with _WF_CANCEL_LOCK:
            _WF_CANCEL_EVENTS.pop(wf_id, None)
        # 更新 last_run + run_count
        try:
            from backend.services.agent_service.workflows_db import WorkflowsDB
            WorkflowsDB().mark_workflow_run(wf_id, datetime.now().isoformat())
        except Exception:
            pass


def _push_workflow_result(owner_id: str, workflow_name: str, results: dict):
    """cron 执行成功后推送结果通知（复用 notifier）"""
    try:
        from backend.services.radar_service.notifier import send_alert
        send_alert(
            owner_id=owner_id,
            keyword=workflow_name,
            platform="workflow",
            risk_level=2,
            risk_class="medium",
            core_issue=f"定时任务「{workflow_name}」执行完成",
            report=f"本次执行结果摘要：{len(results)} 个步骤完成",
            urls=[],
            post_count=1,
        )
    except Exception as e:
        logger.warning(f"[Workflow] 推送结果失败: {e}")
```

## 五、工作流取消 API（★ R-012 修订：路径 + owner 校验）

```python
# backend/services/agent_service/api.py 新增（在 cancel_agent_stream 之后）

@router.post("/api/agent/workflows/{workflow_id}/cancel")
async def cancel_workflow_endpoint(
    workflow_id: int,
    current_user: Dict = Depends(require_registered_user),
):
    """取消正在执行的工作流（仅本人工作流）。

    ★ R-012 修订：
    - 路径前缀为 /api/agent（与现有 api.py 一致）
    - 必须按 owner_id 校验，用户 A 不能取消用户 B 的工作流
    """
    owner_id = str(current_user["id"])

    # 1. 查工作流并校验权限
    from backend.services.agent_service.workflows_db import WorkflowsDB
    db = WorkflowsDB()
    wf = db.get_by_id(workflow_id)
    if not wf:
        raise HTTPException(status_code=404, detail={
            "code": 404, "msg": "工作流不存在"
        })
    if wf["owner_id"] != owner_id and wf["owner_id"] != "system":
        raise HTTPException(status_code=403, detail={
            "code": 403, "msg": "无权操作他人工作流"
        })

    # 2. 触发取消信号
    from backend.services.radar_service.scheduler import (
        _WF_CANCEL_EVENTS, _WF_CANCEL_LOCK
    )
    with _WF_CANCEL_LOCK:
        event = _WF_CANCEL_EVENTS.get(workflow_id)
        if event:
            event.set()
            return {"code": 200, "data": {"cancelled": True}}
    return {"code": 404, "data": {
        "cancelled": False, "message": "工作流未在运行"
    }}


# Admin 取消任意用户工作流（管理员特权）
# 路由在 admin_service/api.py（见 P7）
```

## 七、验收标准
