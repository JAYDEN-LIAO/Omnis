# P1 — 统一工作流存储 + 组合工具动态注册

> 依赖：P0（原子工具层）
> 目标：创建统一的 `workflows` 表（preset + auto_template + user 三种来源），
>       预置组合工具以 DB seed 形式写入，启动时动态注册为 Agent @tool()
> 核心决策：组合工具、自动模板（P2）、用户工作流（P4）三者共用同一张 `workflows` 表，仅以 `source` 字段区分

## 一、统一 workflows 表

```sql
-- workflows_db.py 建表
CREATE TABLE IF NOT EXISTS workflows (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    owner_id TEXT NOT NULL DEFAULT '',       -- 'system' = 预置组合工具
    name TEXT NOT NULL,
    description TEXT DEFAULT '',
    steps_json TEXT NOT NULL,                -- [{step, tool, params, criticality, description, why}]
    constraints_template TEXT DEFAULT '{}',
    source TEXT NOT NULL DEFAULT 'user',     -- 'preset' | 'auto_template' | 'user'
    schedule_cron TEXT DEFAULT '',
    schedule_desc TEXT DEFAULT '',
    enabled INTEGER DEFAULT 1,
    hit_count INTEGER DEFAULT 0,
    last_hit TEXT DEFAULT '',
    last_run TEXT DEFAULT '',
    run_count INTEGER DEFAULT 0,
    is_composite_tool INTEGER DEFAULT 0,     -- 1 = 注册为 Agent Function Calling 工具
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_wf_owner ON workflows(owner_id);
CREATE INDEX IF NOT EXISTS idx_wf_source ON workflows(source);
CREATE INDEX IF NOT EXISTS idx_wf_enabled ON workflows(enabled);
CREATE INDEX IF NOT EXISTS idx_wf_composite ON workflows(is_composite_tool);
```

**三种来源，一张表**：

| source | 谁创建 | is_composite_tool | 触发方式 | 编辑权限 |
|--------|-------|-------------------|---------|---------|
| `preset` | 开发者（DB seed） | 1 | Agent Function Calling | Admin only |
| `auto_template` | Agent 自动积累 | 0 | phase_plan 模板匹配（P2） | 不可编辑 |
| `user` | 用户自然语言创建 | 0（可升级为 1） | cron / 手动 / Agent 调用（P4） | 用户可编辑 |

## 三、动态注册（tools/composite.py）

```python
# backend/services/agent_service/tools/composite.py
"""组合工具动态注册（I 组）

启动时从 workflows 表读取所有 is_composite_tool=1 AND enabled=1 的条目，
动态注册为 Agent Function Calling 工具。
"""
import json
from typing import Optional

from backend.core.logger import get_logger

from ._base import tool, _REGISTRY, get_tool
from ._owner import with_owner

logger = get_logger("agent.tools.composite")


def register_composite_tools():
    """启动时调用：从 workflows 表读取 → 动态注册 @tool()

    幂等：_REGISTRY 中已存在的同名工具跳过（重启 / 新增 preset 不会抛 ValueError）。
    """
    from backend.services.agent_service.workflows_db import WorkflowsDB
    db = WorkflowsDB()
    presets = db.get_composite_tools()
    registered = 0
    for wf in presets:
        if _register_single(wf):
            registered += 1
    logger.info(f"[Composite] 动态注册完成：{registered} 个新工具")


def _register_single(wf: dict) -> bool:
    """注册单个组合工具。返回 True 表示新增，False 表示已存在跳过。"""
    if wf["name"] in _REGISTRY:
        return False

    steps = json.loads(wf["steps_json"])
    params_schema = _extract_params_from_steps(steps)

    @tool(
        name=wf["name"],
        description=wf["description"],
        parameters=params_schema,
        group="composite",
    )
    @with_owner
    async def composite_tool(_owner_id: str = "", **kwargs):
        """通用组合工具执行器。

        启动一个 inline EventLoop 跑 execute_workflow_steps；
        cancel_event 从 contextvar 取（Agent SSE 流中存在，cron 触发时为 None）。
        """
        from backend.services.agent_service.workflows_db import execute_workflow_steps
        from ._base import get_current_cancel_event
        cancel_event = get_current_cancel_event()
        return await execute_workflow_steps(
            steps=steps, initial_params=kwargs,
            owner_id=_owner_id, cancel_event=cancel_event,
        )

    composite_tool.__name__ = wf["name"]
    return True


def _extract_params_from_steps(steps: list) -> dict:
    """从 steps 中提取 {{param_name}} 占位符，生成 OpenAI function calling schema。

    只识别 step_N 之外的占位符作为工具入参；step_N_result 视为内部引用。
    """
    import re
    PLACEHOLDER_RE = re.compile(r"^\s*\{\{\s*([a-zA-Z_][a-zA-Z0-9_]*)\s*\}\}\s*$")
    STEP_REF_RE = re.compile(r"^\s*step_(\d+)(?:\..*)?\s*$")

    params = {"type": "object", "properties": {}, "required": []}
    seen = set()

    for step in steps:
        for key, val in step.get("params", {}).items():
            if not isinstance(val, str):
                continue
            m = PLACEHOLDER_RE.match(val)
            if not m:
                continue
            pname = m.group(1)
            # 排除 step 引用
            if STEP_REF_RE.match(pname):
                continue
            if pname in seen:
                continue
            seen.add(pname)
            params["properties"][pname] = {
                "type": "string",
                "description": f"参数: {pname}",
            }
            if step.get("criticality") == "critical":
                params["required"].append(pname)
    return params
```

## 四、execute_workflow_steps — 统一执行器（含 R-001/R-002/R-004 修订）

```python
# workflows_db.py
import asyncio
import json
import re
import time
from datetime import datetime
from typing import Any, Dict, List, Optional

from backend.core.database import get_db_connection
from backend.core.logger import get_logger
from backend.services.agent_service.tools._base import get_tool

logger = get_logger("agent.workflows")


# ★ R-004 修订：用正则提取 step_N 中的整数 N
_STEP_IDX_RE = re.compile(r"^step_(\d+)$")


def _resolve_step_index(token: str) -> Optional[int]:
    m = _STEP_IDX_RE.match(token.strip())
    return int(m.group(1)) if m else None


def _resolve_params(params: dict, results: dict) -> dict:
    """解析 {{step_N_result}} 和 {{step_N_result.field.path}} 模板变量。

    只对 step_N 形式的占位符执行替换；其他占位符（如 {{keyword}}）原样保留，
    由 execute_workflow_steps 上层用 initial_params 覆盖。
    """
    resolved = {}
    for key, value in params.items():
        if not (isinstance(value, str) and value.startswith("{{") and value.endswith("}}")):
            resolved[key] = value
            continue

        ref = value[2:-2].strip()
        parts = ref.split(".")
        idx = _resolve_step_index(parts[0])
        if idx is None:
            # 非 step 占位符，保留原样（initial_params 会覆盖）
            resolved[key] = value
            continue

        val = results.get(f"step_{idx}")
        if val is None:
            resolved[key] = {}
            continue

        # 逐层下钻字段
        for p in parts[1:]:
            if isinstance(val, dict):
                val = val.get(p)
            elif isinstance(val, str):
                try:
                    val = json.loads(val).get(p)
                except Exception:
                    val = None
            else:
                val = None
            if val is None:
                break
        resolved[key] = val
    return resolved


async def execute_workflow_steps(
    steps: list,
    initial_params: dict,
    owner_id: str,
    cancel_event: Optional[asyncio.Event] = None,
    workflow_id: Optional[int] = None,
    workflow_name: str = "",
) -> str:
    """统一步骤执行器。

    Args:
        steps: 工作流步骤列表，每项含 {tool, params, criticality, ...}
        initial_params: 入口参数（来自用户输入或 cron 触发）
        owner_id: 当前 owner（决定数据隔离，**必传**）
        cancel_event: 取消事件（Agent SSE 时存在，cron 时 None）
        workflow_id: 来自 DB 工作流时记录 id 用于执行日志关联
        workflow_name: 工作流名（写入执行日志）

    Returns:
        JSON 字符串：成功 {"success": true, "results": {...}}；
        失败 {"error": ..., "step": N}；取消 {"error": "cancelled", "step": N}
    """
    results: Dict[str, Any] = {}
    if not owner_id:
        # 安全网：缺 owner_id 拒绝执行（防止组合工具漏 owner 上下文）
        return json.dumps({"error": "owner_id 缺失", "step": 0}, ensure_ascii=False)

    start_ts = time.time()
    failed_step = 0
    status = "success"

    for i, step in enumerate(steps):
        # 取消检查
        if cancel_event and cancel_event.is_set():
            status = "cancelled"
            failed_step = i + 1
            _log_workflow_execution(
                workflow_id, workflow_name, owner_id, "cancelled",
                time.time() - start_ts, len(steps), failed_step, "用户取消",
            )
            return json.dumps({
                "error": "cancelled", "step": i + 1, "message": "用户取消"
            }, ensure_ascii=False)

        tool_name = step["tool"]
        params = _resolve_params(step.get("params", {}), results)
        # initial_params 覆盖（用户入口参数优先级最高）
        for k, v in initial_params.items():
            if k not in params:
                params[k] = v
        params["_owner_id"] = owner_id  # 强制注入

        # ★ R-001 关键修正：get_tool 返回 {func, schema, group, streamable} dict
        meta = get_tool(tool_name)
        if not meta or "func" not in meta:
            return json.dumps({
                "error": f"工具 {tool_name} 不存在或未注册", "step": i + 1
            }, ensure_ascii=False)
        tool_func = meta["func"]

        criticality = step.get("criticality", "required")
        max_retries = {"critical": 1, "required": 2, "optional": 0}.get(criticality, 1)

        last_error = None
        for attempt in range(max_retries + 1):
            try:
                # ★ R-002 兼容 sync / async 函数
                if asyncio.iscoroutinefunction(tool_func):
                    result = await tool_func(**params)
                else:
                    result = await asyncio.to_thread(tool_func, **params)
                results[f"step_{i+1}"] = result
                last_error = None
                break
            except Exception as e:
                last_error = e
                logger.warning(
                    f"[Workflow] step {i+1} ({tool_name}) "
                    f"重试 {attempt+1}/{max_retries+1} 失败: {e}"
                )
                if attempt < max_retries:
                    continue
                # 重试耗尽
                if criticality == "critical":
                    status = "failed"
                    failed_step = i + 1
                    _log_workflow_execution(
                        workflow_id, workflow_name, owner_id, "failed",
                        time.time() - start_ts, len(steps), failed_step, str(e),
                    )
                    return json.dumps({
                        "error": str(e), "step": i + 1,
                        "message": "关键步骤失败，执行终止"
                    }, ensure_ascii=False)
                # required/optional 降级
                results[f"step_{i+1}"] = json.dumps({
                    "skipped": True, "reason": str(e)
                }, ensure_ascii=False)
                last_error = None
                break

    duration = time.time() - start_ts
    _log_workflow_execution(
        workflow_id, workflow_name, owner_id, status,
        duration, len(steps), failed_step,
        str(last_error) if last_error else "",
    )
    return json.dumps({"success": True, "results": results}, ensure_ascii=False)


def _log_workflow_execution(
    workflow_id: Optional[int], workflow_name: str, owner_id: str,
    status: str, duration: float, total_steps: int,
    failed_step: int, error_message: str,
):
    """执行日志落库（失败时静默，避免影响主流程）"""
    try:
        from backend.services.admin_service.agent_health_db import AgentHealthDB
        db = AgentHealthDB()
        db.insert_workflow_log(
            workflow_id=workflow_id or 0,
            workflow_name=workflow_name or "",
            owner_id=owner_id or "",
            status=status,
            duration_seconds=duration,
            total_steps=total_steps,
            failed_step=failed_step,
            error_message=error_message[:500] if error_message else "",
        )
    except Exception as e:
        logger.warning(f"[Workflow] 写执行日志失败: {e}")
```

```python
# workflows_db.py — WorkflowsDB 类

class WorkflowsDB:
    """统一工作流 CRUD（preset + auto_template + user 共用）"""

    def __init__(self, db_path: str = None):
        from backend.core.database import get_db_connection
        self.db_path = db_path
        self._init_db()

    def _get_conn(self):
        from backend.core.database import get_db_connection
        return get_db_connection(self.db_path) if self.db_path else get_db_connection()

    def _init_db(self):
        with self._get_conn() as conn:
            c = conn.cursor()
            c.execute("""
                CREATE TABLE IF NOT EXISTS workflows (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    owner_id TEXT NOT NULL DEFAULT '',
                    name TEXT NOT NULL,
                    description TEXT DEFAULT '',
                    steps_json TEXT NOT NULL,
                    constraints_template TEXT DEFAULT '{}',
                    source TEXT NOT NULL DEFAULT 'user',
                    schedule_cron TEXT DEFAULT '',
                    schedule_desc TEXT DEFAULT '',
                    enabled INTEGER DEFAULT 1,
                    hit_count INTEGER DEFAULT 0,
                    last_hit TEXT DEFAULT '',
                    last_run TEXT DEFAULT '',
                    run_count INTEGER DEFAULT 0,
                    is_composite_tool INTEGER DEFAULT 0,
                    created_at TEXT NOT NULL,
                    updated_at TEXT NOT NULL
                )
            """)
            for ddl in [
                "CREATE INDEX IF NOT EXISTS idx_wf_owner ON workflows(owner_id)",
                "CREATE INDEX IF NOT EXISTS idx_wf_source ON workflows(source)",
                "CREATE INDEX IF NOT EXISTS idx_wf_enabled ON workflows(enabled)",
                "CREATE INDEX IF NOT EXISTS idx_wf_composite ON workflows(is_composite_tool)",
            ]:
                c.execute(ddl)
            conn.commit()

    def query(self, sql: str, params: list = None) -> List[Dict]:
        import sqlite3
        with self._get_conn() as conn:
            conn.row_factory = sqlite3.Row
            rows = conn.execute(sql, params or []).fetchall()
            return [dict(r) for r in rows]

    def get_composite_tools(self) -> List[Dict]:
        return self.query(
            "SELECT * FROM workflows WHERE is_composite_tool=1 AND enabled=1"
        )

    def get_by_id(self, wf_id: int) -> Optional[Dict]:
        rows = self.query("SELECT * FROM workflows WHERE id=?", [wf_id])
        return rows[0] if rows else None

    def upsert_auto_template(
        self, owner_id: str, name: str, steps_json: str,
        constraints_template: str = "{}",
    ) -> int:
        """累积 auto_template：已存在同 owner + 同 name 时 hit_count++，否则 INSERT。"""
        existing = self.query(
            "SELECT id, hit_count FROM workflows "
            "WHERE owner_id=? AND name=? AND source='auto_template'",
            [owner_id, name],
        )
        now = datetime.now().isoformat()
        with self._get_conn() as conn:
            c = conn.cursor()
            if existing:
                wf_id = existing[0]["id"]
                c.execute("""
                    UPDATE workflows
                    SET hit_count = hit_count + 1,
                        last_hit = ?,
                        enabled = 1,
                        updated_at = ?
                    WHERE id = ?
                """, [now, now, wf_id])
            else:
                c.execute("""
                    INSERT INTO workflows
                    (owner_id, name, steps_json, constraints_template, source,
                     enabled, hit_count, last_hit, is_composite_tool, created_at, updated_at)
                    VALUES (?, ?, ?, ?, 'auto_template', 1, 1, ?, 0, ?, ?)
                """, [owner_id, name, steps_json, constraints_template,
                      now, now, now])
                wf_id = c.lastrowid
            conn.commit()
        return wf_id

    def insert_workflow(
        self, owner_id: str, name: str, steps_json: str,
        constraints_template: str = "{}",
        source: str = "user",
        schedule_cron: str = "",
        schedule_desc: str = "",
        is_composite_tool: int = 0,
    ) -> int:
        now = datetime.now().isoformat()
        with self._get_conn() as conn:
            c = conn.cursor()
            c.execute("""
                INSERT INTO workflows
                (owner_id, name, steps_json, constraints_template, source,
                 schedule_cron, schedule_desc, is_composite_tool,
                 enabled, created_at, updated_at)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, 1, ?, ?)
            """, [owner_id, name, steps_json, constraints_template, source,
                  schedule_cron, schedule_desc, is_composite_tool,
                  now, now])
            conn.commit()
            return c.lastrowid

    def set_enabled(self, wf_id: int, enabled: bool):
        with self._get_conn() as conn:
            c = conn.cursor()
            c.execute(
                "UPDATE workflows SET enabled=?, updated_at=? WHERE id=?",
                [1 if enabled else 0, datetime.now().isoformat(), wf_id],
            )
            conn.commit()

    def update(self, wf_id: int, **fields):
        """白名单字段更新（避免 SQL 注入）"""
        allowed = {"name", "description", "steps_json", "constraints_template",
                   "schedule_cron", "schedule_desc", "source", "enabled",
                   "is_composite_tool"}
        sets = []
        params = []
        for k, v in fields.items():
            if k in allowed:
                sets.append(f"{k}=?")
                params.append(v)
        if not sets:
            return
        sets.append("updated_at=?")
        params.append(datetime.now().isoformat())
        params.append(wf_id)
        with self._get_conn() as conn:
            c = conn.cursor()
            c.execute(f"UPDATE workflows SET {', '.join(sets)} WHERE id=?", params)
            conn.commit()

    def delete(self, wf_id: int):
        with self._get_conn() as conn:
            c = conn.cursor()
            c.execute("DELETE FROM workflows WHERE id=?", [wf_id])
            conn.commit()

    def list_due_user_workflows(self) -> List[Dict]:
        """查待执行的定时任务（cron 字段非空 + enabled）"""
        return self.query(
            "SELECT * FROM workflows WHERE source='user' "
            "AND schedule_cron != '' AND enabled=1"
        )

    def mark_workflow_run(self, wf_id: int, last_run_iso: str):
        """更新 last_run + run_count++"""
        with self._get_conn() as conn:
            c = conn.cursor()
            c.execute("""
                UPDATE workflows
                SET last_run=?, run_count=run_count+1, updated_at=?
                WHERE id=?
            """, [last_run_iso, datetime.now().isoformat(), wf_id])
            conn.commit()
```
